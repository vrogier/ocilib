114inline ostring Vector::ToString() const
115{
116 if (!IsNull())
117 {
118 unsigned int size{};
119
120 core::Check(OCI_VectorToText(*this, &size, nullptr));
121
122 ostring result;
123
124 result.resize(static_cast<size_t>(size));
125
126 core::Check(OCI_VectorToText(*this, &size, const_cast<otext*>(result.data())));
127
128 result.resize(static_cast<size_t>(size));
129
130 return result;
131 }
132
133 return OCI_STRING_NULL;
